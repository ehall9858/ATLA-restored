##################################################
# Avatar Sub-Bending
##################################################
#
# Lets the Avatar learn the specialist arts of the elements they command.
#
# Why this file exists: every ordinary route to a sub-bending trait tests for
# the plain element trait (earthbender/firebender/waterbender/airbender) or for
# a condition no Avatar meets (firelord, sandbender culture, society rank, or
# being a child with an educator). The Avatar carries avatar_<element>bender
# instead, so all of those routes miss them and they could never learn a single
# specialist art.
#
# Two ways in:
#   1. The journey. Meeting the sandbenders during the earth leg of the Avatar
#      journey (new_avatar_journey.201 -> atla_avatar_sub.1) can end in the
#      sandbending art, either at once or once earthbending has been learned.
#   2. Insight. A rare, long-interval moment of understanding for an Avatar who
#      has already pushed an element to master tier (atla_avatar_sub.10/20/30/40).
#      Deliberately not guaranteed - many Avatars will never find one.
#
# Nothing here can fire for an unrevealed Avatar child: every trigger requires
# is_revealed_avatar_trigger.
##################################################

namespace = atla_avatar_sub


########################
# 1. The Way of the Sand  (journey: met the sandbenders)
########################

narrative_event = {
	id = atla_avatar_sub.1
	title = EVTTITLEatla_avatar_sub.1
	desc = EVTDESCatla_avatar_sub.1
	picture = GFX_evt_sandbender
	is_triggered_only = yes

	# Already able to move earth - try to learn their art while you are here.
	option = {
		name = EVTOPTAatla_avatar_sub.1
		trigger = {
			OR = {
				trait = earthbender
				trait = avatar_earthbender
			}
			NOT = {
				trait = sandbender
			}
		}
		random_list = {
			45 = {
				add_trait = sandbender
				prestige = 50
				modifier = {
					factor = 1.5
					trait = patient
				}
				modifier = {
					factor = 1.5
					trait = diligent
				}
			}
			55 = {
				prestige = 10
				modifier = {
					factor = 1.5
					trait = slothful
				}
				modifier = {
					factor = 1.25
					trait = content
				}
			}
		}
		character_event = {
			id = avatar.159
			days = 10
			random = 20
		}
	}

	# Cannot move earth yet - but the dunes and their people are not forgotten.
	option = {
		name = EVTOPTBatla_avatar_sub.1
		trigger = {
			NOR = {
				trait = earthbender
				trait = avatar_earthbender
			}
		}
		set_character_flag = avatar_studied_sandbenders
		character_event = {
			id = avatar.159
			days = 10
			random = 20
		}
	}

	# Move on.
	option = {
		name = EVTOPTCatla_avatar_sub.1
		character_event = {
			id = avatar.159
			days = 10
			random = 20
		}
	}
}
