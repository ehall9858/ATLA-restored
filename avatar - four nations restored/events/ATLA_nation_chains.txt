namespace = NAFC

# Hidden yearly starter. The on_action slot is guaranteed, and this trigger
# filters it down to adult Fire Nation rulers. The chance keeps the chain rare.
character_event = {
	id = NAFC.100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_landed = yes
		higher_real_tier_than = BARON
		is_adult = yes
		is_incapable = no
		NOT = { prisoner = yes }
		culture_group = fire_nation_cultures
		NOT = { has_character_flag = nafc_fire_chain_active }
		NOT = { has_character_flag = nafc_fire_chain_done }
	}

	immediate = {
		random = {
			chance = 20
			hidden_tooltip = {
				set_character_flag = nafc_fire_chain_active
				character_event = { id = NAFC.101 days = 3 }
			}
		}
	}
}
