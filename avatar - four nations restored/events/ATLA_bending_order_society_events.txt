# ---------------------------------------------------------------- recruitment

# The Order writes to a waterbender it wants.
letter_event = {
	id = atla_bending_society.10
	desc = EVTDESC_ABS_10
	border = GFX_event_letter_frame_religion
	is_triggered_only = yes

	trigger = {
		is_in_society = no
		is_adult = yes
		trait = waterbender
	}

	immediate = {
		set_character_flag = society_join_block
	}

	option = {
		name = EVTOPTA_ABS_10
		join_society = waterbending_order
		clr_character_flag = society_join_block
		piety = 25
	}

	option = {
		name = EVTOPTB_ABS_10
		clr_character_flag = society_join_block
		trigger = { ai = no }
	}
}
