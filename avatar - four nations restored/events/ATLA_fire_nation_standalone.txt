namespace = NAFS

# Hidden yearly dispatcher for the Fire Nation standalone events. The on_action
# slot is guaranteed; this trigger and chance keep the events rare.
character_event = {
	id = NAFS.99
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_landed = yes
		higher_real_tier_than = BARON
		is_adult = yes
		is_incapable = no
		NOT = { prisoner = yes }
		culture_group = fire_nation_cultures
		OR = {
			NOT = { has_character_flag = nafs_fire_dragon_done }
			AND = {
				year = 830
				NOT = { has_character_flag = nafs_fire_opera_done }
			}
			AND = {
				year = 850
				NOT = { has_character_flag = nafs_fire_tax_done }
			}
			AND = {
				year = 918
				NOT = { has_character_flag = nafs_fire_balloon_done }
			}
			NOT = { has_character_flag = nafs_fire_festival_done }
		}
	}

	immediate = {
		random = {
			chance = 20
			random_list = {
				100 = {
					trigger = { NOT = { has_character_flag = nafs_fire_dragon_done } }
					character_event = { id = NAFS.100 }
				}
				100 = {
					trigger = {
						year = 918
						NOT = { has_character_flag = nafs_fire_balloon_done }
					}
					character_event = { id = NAFS.101 }
				}
				100 = {
					trigger = {
						year = 830
						NOT = { has_character_flag = nafs_fire_opera_done }
					}
					character_event = { id = NAFS.102 }
				}
				100 = {
					trigger = {
						year = 850
						NOT = { has_character_flag = nafs_fire_tax_done }
					}
					character_event = { id = NAFS.103 }
				}
				100 = {
					trigger = { NOT = { has_character_flag = nafs_fire_festival_done } }
					character_event = { id = NAFS.104 }
				}
			}
		}
	}
}

# The Dragon Keeper's Request
character_event = {
	id = NAFS.100
	desc = NAFS.100.desc
	picture = GFX_evt_dragon
	border = GFX_event_normal_frame_religion
	is_triggered_only = yes
	only_playable = yes

	trigger = {
		is_landed = yes
		higher_real_tier_than = BARON
		is_adult = yes
		is_incapable = no
		NOT = { prisoner = yes }
		culture_group = fire_nation_cultures
		NOT = { has_character_flag = nafs_fire_dragon_done }
	}

	option = {
		name = NAFS.100.a
		ai_chance = {
			factor = 40
			modifier = {
				factor = 2
				trait = kind
			}
			modifier = {
				factor = 1.5
				trait = zealous
			}
		}
		piety = 30
		prestige = 10
		if = {
			limit = { NOT = { trait = kind } }
			hidden_tooltip = {
				remove_trait = cruel
				remove_trait = envious
			}
			add_trait = kind
		}
		hidden_tooltip = { set_character_flag = nafs_fire_dragon_done }
	}

	option = {
		name = NAFS.100.b
		ai_chance = {
			factor = 35
			modifier = {
				factor = 2
				trait = proud
			}
			modifier = {
				factor = 1.5
				trait = ambitious
			}
		}
		prestige = 40
		piety = -10
		if = {
			limit = { NOT = { trait = proud } }
			hidden_tooltip = { remove_trait = humble }
			add_trait = proud
		}
		hidden_tooltip = { set_character_flag = nafs_fire_dragon_done }
	}

	option = {
		name = NAFS.100.c
		ai_chance = {
			factor = 30
			modifier = {
				factor = 2
				trait = zealous
			}
			modifier = {
				factor = 1.5
				trait = patient
			}
		}
		piety = 25
		prestige = 5
		if = {
			limit = { NOT = { trait = zealous } }
			hidden_tooltip = { remove_trait = cynical }
			add_trait = zealous
		}
		hidden_tooltip = { set_character_flag = nafs_fire_dragon_done }
	}
}
