#
#  atla_bending_missions.1    Waterbending Order - mission tombola

#
#  atla_bending_missions.10   water - Tend the Sick

#  atla_bending_missions.12   water - Bring a New Voice
#
#  atla_bending_missions.20   earth - Raise the Wall
#  atla_bending_missions.21   earth - Stand the Trial

#
#  atla_bending_missions.30   water - the work is done
#  atla_bending_missions.31   earth - the work is done
#  atla_bending_missions.32   water - the task is set aside
#  atla_bending_missions.33   earth - the task is set aside
#
# The tombolas are hooked into on_society_bi_yearly_pulse. Missions are for
# the player: the AI branch is switched off so that an AI member cannot take
# a quest it will never finish and block its own future missions. Tasks are
# completed from decisions/ATLA_bending_order_missions_decisions.txt.
########################################################################

namespace = atla_bending_missions

# ------------------------------------------------------------------ tombolas

character_event = {
	id = atla_bending_missions.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		ai = no
		prisoner = no
		society_member_of = waterbending_order
		is_society_grandmaster = no
		has_any_quest = no
		NOT = { is_inaccessible_trigger = yes }
		NOT = { has_character_modifier = quest_cooldown_timer }
	}

	immediate = {
		random_list = {
			30 = {
				society_quest_event = { id = atla_bending_missions.10 }
			}
			30 = {
				society_quest_event = { id = atla_bending_missions.11 }
			}
			30 = {
				society_quest_event = { id = atla_bending_missions.12 }
			}
			10 = { # the Order asks nothing of you this season
			}
		}
	}
}

# --------------------------------------------------------------- water offers

# Tend the sick of your own household.
society_quest_event = {
	id = atla_bending_missions.10
	desc = EVTDESC_ABM_10
	border = GFX_event_letter_frame_religion
	is_triggered_only = yes

	trigger = {
		any_courtier = {
			OR = {
				trait = wounded
				trait = stressed
				trait = depressed
			}
		}
	}

	option = {
		name = ACCEPT
		set_quest = quest_wbo_tend_the_sick
	}

	option = {
		name = DECLINE
		hidden_effect = {
			add_character_modifier = {
				name = quest_cooldown_timer
				hidden = yes
				days = 1000
			}
		}
	}
}
