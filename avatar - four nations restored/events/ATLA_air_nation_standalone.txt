namespace = NAFS

# Hidden yearly dispatcher for the Air Nomad standalone events. The on_action
# slot is guaranteed; this trigger and chance keep the events rare.
character_event = {
	id = NAFS.499
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_landed = yes
		higher_real_tier_than = BARON
		is_adult = yes
		is_incapable = no
		NOT = { prisoner = yes }
		culture_group = air_nomad_cultures
		religion_group = air_religions
		NOT = { culture = colonial_air_fire }
		NOT = { culture = colonial_air_earth }
		NOT = { culture = colonial_air_water }
		OR = {
			NOT = { has_character_flag = nafs_air_poachers_done }
			NOT = { has_character_flag = nafs_air_orchard_done }
			NOT = { has_character_flag = nafs_air_warlord_done }
			AND = {
				NOT = { year = 830 }
				NOT = { has_character_flag = nafs_air_lostbender_done }
			}
			NOT = { has_character_flag = nafs_air_glider_done }
			NOT = { has_character_flag = nafs_air_hermit_done }
		}
	}

	immediate = {
		random = {
			chance = 20
			random_list = {
				100 = {
					trigger = { NOT = { has_character_flag = nafs_air_poachers_done } }
					character_event = { id = NAFS.400 }
				}
				100 = {
					trigger = { NOT = { has_character_flag = nafs_air_orchard_done } }
					character_event = { id = NAFS.401 }
				}
				100 = {
					trigger = { NOT = { has_character_flag = nafs_air_warlord_done } }
					character_event = { id = NAFS.402 }
				}
				100 = {
					trigger = {
						NOT = { year = 830 }
						NOT = { has_character_flag = nafs_air_lostbender_done }
					}
					character_event = { id = NAFS.403 }
				}
				100 = {
					trigger = { NOT = { has_character_flag = nafs_air_glider_done } }
					character_event = { id = NAFS.404 }
				}
				100 = {
					trigger = { NOT = { has_character_flag = nafs_air_hermit_done } }
					character_event = { id = NAFS.405 }
				}
			}
		}
	}
}

# The Poachers' Calf
character_event = {
	id = NAFS.400
	desc = NAFS.400.desc
	picture = GFX_evt_air_west
	border = GFX_event_normal_frame_war
	is_triggered_only = yes
	only_playable = yes

	trigger = {
		is_landed = yes
		higher_real_tier_than = BARON
		is_adult = yes
		is_incapable = no
		NOT = { prisoner = yes }
		culture_group = air_nomad_cultures
		religion_group = air_religions
		NOT = { culture = colonial_air_fire }
		NOT = { culture = colonial_air_earth }
		NOT = { culture = colonial_air_water }
		NOT = { has_character_flag = nafs_air_poachers_done }
	}

	option = {
		name = NAFS.400.a
		ai_chance = {
			factor = 40
			modifier = {
				factor = 2
				trait = kind
			}
			modifier = {
				factor = 1.5
				trait = charitable
			}
			modifier = {
				factor = 1.5
				trait = just
			}
		}
		wealth = -30
		prestige = 20
		piety = 20
		if = {
			limit = { NOT = { trait = kind } }
			hidden_tooltip = {
				remove_trait = cruel
				remove_trait = envious
			}
			add_trait = kind
		}
		hidden_tooltip = { set_character_flag = nafs_air_poachers_done }
	}

	option = {
		name = NAFS.400.b
		ai_chance = {
			factor = 35
			modifier = {
				factor = 2
				trait = brave
			}
			modifier = {
				factor = 1.5
				trait = ambitious
			}
		}
		prestige = 30
		change_martial = 1
		if = {
			limit = { NOT = { trait = brave } }
			hidden_tooltip = { remove_trait = craven }
			add_trait = brave
		}
		hidden_tooltip = { set_character_flag = nafs_air_poachers_done }
	}

	option = {
		name = NAFS.400.c
		ai_chance = {
			factor = 25
			modifier = {
				factor = 2
				trait = patient
			}
			modifier = {
				factor = 1.5
				trait = just
			}
		}
		prestige = 10
		change_stewardship = 1
		if = {
			limit = { NOT = { trait = patient } }
			hidden_tooltip = { remove_trait = wroth }
			add_trait = patient
		}
		hidden_tooltip = { set_character_flag = nafs_air_poachers_done }
	}
}
