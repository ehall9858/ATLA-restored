namespace = avatar_child

# Childhood flavor for the unrevealed Avatar (ages 0-15). Checked yearly;
# the child gets one small strange-and-wonderful moment per year. NOTHING
# here may reveal them: no avatar trait, no reveal flags, no titles, and
# no death - the unrevealedavatar trait is keeping them alive for a reason.
#
# Every desc is written in the child's own voice ("I", "my"), because these
# fire on the child themselves. Losing options roll on percentages so the
# same childhood never plays out the same way twice.
character_event = {
    id = avatar_child.0
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_character_flag = avatar_unrevealed
        NOT = { has_character_flag = avatar_revealed }
    }
    immediate = {
        # Toddlers: small household wonders.
        if = {
            limit = { NOT = { age >= 6 } }
            random_list = {
                10 = { character_event = { id = avatar_child.1 } }
                10 = { character_event = { id = avatar_child.2 } }
                10 = { character_event = { id = avatar_child.3 } }
                10 = { character_event = { id = avatar_child.10 } }
                10 = { character_event = { id = avatar_child.11 } }
                10 = { character_event = { id = avatar_child.12 } }
                10 = { character_event = { id = avatar_child.13 } }
            }
        }
        # Children: talents and dreams.
        if = {
            limit = {
                age >= 6
                NOT = { age >= 12 }
            }
            random_list = {
                10 = { character_event = { id = avatar_child.4 } }
                10 = { character_event = { id = avatar_child.5 } }
                10 = { character_event = { id = avatar_child.6 } }
                10 = { character_event = { id = avatar_child.14 } }
                10 = { character_event = { id = avatar_child.15 } }
                10 = { character_event = { id = avatar_child.16 } }
                10 = { character_event = { id = avatar_child.17 } }
            }
        }
        # Almost grown: the world starts noticing.
        if = {
            limit = {
                age >= 12
                NOT = { age >= 16 }
            }
            random_list = {
                10 = { character_event = { id = avatar_child.7 } }
                10 = { character_event = { id = avatar_child.8 } }
                10 = { character_event = { id = avatar_child.9 } }
                10 = { character_event = { id = avatar_child.18 } }
                10 = { character_event = { id = avatar_child.19 } }
                10 = { character_event = { id = avatar_child.20 } }
                10 = { character_event = { id = avatar_child.21 } }
            }
        }
    }
}
