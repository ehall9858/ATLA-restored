namespace = NAFS

# Hidden yearly dispatcher for the Water Tribe standalone events. The on_action
# slot is guaranteed; this trigger and chance keep the events rare.
character_event = {
	id = NAFS.199
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_landed = yes
		higher_real_tier_than = BARON
		is_adult = yes
		is_incapable = no
		NOT = { prisoner = yes }
		OR = {
			culture = northern_water_culture
			culture = southern_water_culture
		}
		OR = {
			NOT = { has_character_flag = nafs_water_betrothal_done }
			NOT = { has_character_flag = nafs_water_healing_done }
			AND = {
				any_realm_province = { is_coastal = yes }
				NOT = { has_character_flag = nafs_water_iceberg_done }
			}
			AND = {
				culture = southern_water_culture
				year = 830
				e_fire_nation = { has_holder = yes }
				any_realm_province = { is_coastal = yes }
				NOT = { has_character_flag = nafs_water_raiders_done }
			}
			NOT = { has_character_flag = nafs_water_scroll_done }
			AND = {
				culture = northern_water_culture
				any_realm_province = { is_coastal = yes }
				NOT = { has_character_flag = nafs_water_wall_done }
			}
		}
	}

	immediate = {
		random = {
			chance = 20
			random_list = {
				100 = {
					trigger = { NOT = { has_character_flag = nafs_water_betrothal_done } }
					character_event = { id = NAFS.200 }
				}
				100 = {
					trigger = { NOT = { has_character_flag = nafs_water_healing_done } }
					character_event = { id = NAFS.201 }
				}
				100 = {
					trigger = {
						any_realm_province = { is_coastal = yes }
						NOT = { has_character_flag = nafs_water_iceberg_done }
					}
					character_event = { id = NAFS.202 }
				}
				100 = {
					trigger = {
						culture = southern_water_culture
						year = 830
						e_fire_nation = { has_holder = yes }
						any_realm_province = { is_coastal = yes }
						NOT = { has_character_flag = nafs_water_raiders_done }
					}
					character_event = { id = NAFS.203 }
				}
				100 = {
					trigger = { NOT = { has_character_flag = nafs_water_scroll_done } }
					character_event = { id = NAFS.204 }
				}
				100 = {
					trigger = {
						culture = northern_water_culture
						any_realm_province = { is_coastal = yes }
						NOT = { has_character_flag = nafs_water_wall_done }
					}
					character_event = { id = NAFS.205 }

			}
		}
	}
}
