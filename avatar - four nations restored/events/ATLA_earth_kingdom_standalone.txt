namespace = NAFS

# Hidden yearly dispatcher for the Earth Kingdom standalone events. The on_action
# slot is guaranteed; this trigger and chance keep the events rare.
character_event = {
	id = NAFS.299
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_landed = yes
		higher_real_tier_than = BARON
		is_adult = yes
		is_incapable = no
		NOT = { prisoner = yes }
		culture_group = earth_kingdom_cultures
		NOT = { culture = barbarian_culture }
		OR = {
			AND = {
				NOT = { culture = sandbender_culture }
				NOT = { has_character_flag = nafs_earth_magistrate_done }
			}
			AND = {
				NOT = { culture = sandbender_culture }
				NOT = { has_character_flag = nafs_earth_granary_done }
			}
			AND = {
				NOT = { culture = sandbender_culture }
				NOT = { has_character_flag = nafs_earth_wall_done }
			}
			AND = {
				NOT = { culture = sandbender_culture }
				has_global_flag = HYW_0
				NOT = { has_character_flag = nafs_earth_refugees_done }
			}
			AND = {
				OR = {
					culture = sandbender_culture
					capital_scope = {
						OR = {
							region = world_central_desert
							region = world_si_wong
							region = world_si_wong_fringe
						}
					}
				}
				NOT = { has_character_flag = nafs_earth_desert_done }
			}
			AND = {
				NOT = { culture = sandbender_culture }
				has_global_flag = kyoshi_warriors_founded
				OR = {
					culture = earth_islander
					capital_scope = { region = world_earth_islands }
					capital_scope = { region = world_eastern_coast }
				}
				NOT = { has_character_flag = nafs_earth_kyoshi_done }
			}
		}
	}

	immediate = {
		random = {
			chance = 20
			random_list = {
				100 = {
					trigger = {
						NOT = { culture = sandbender_culture }
						NOT = { has_character_flag = nafs_earth_magistrate_done }
					}
					character_event = { id = NAFS.300 }
				}
				100 = {
					trigger = {
						NOT = { culture = sandbender_culture }
						NOT = { has_character_flag = nafs_earth_granary_done }
					}
					character_event = { id = NAFS.301 }
				}
				100 = {
					trigger = {
						NOT = { culture = sandbender_culture }
						NOT = { has_character_flag = nafs_earth_wall_done }
					}
					character_event = { id = NAFS.302 }
				}
				100 = {
					trigger = {
						NOT = { culture = sandbender_culture }
						has_global_flag = HYW_0
						NOT = { has_character_flag = nafs_earth_refugees_done }
					}
					character_event = { id = NAFS.303 }
				}
				100 = {
					trigger = {
						OR = {
							culture = sandbender_culture
							capital_scope = {
								OR = {
									region = world_central_desert
									region = world_si_wong
									region = world_si_wong_fringe
								}
							}
						}
						NOT = { has_character_flag = nafs_earth_desert_done }
					}
					character_event = { id = NAFS.304 }
				}
				100 = {
					trigger = {
						NOT = { culture = sandbender_culture }
						has_global_flag = kyoshi_warriors_founded
						OR = {
							culture = earth_islander
							capital_scope = { region = world_earth_islands }
							capital_scope = { region = world_eastern_coast }
						}
						NOT = { has_character_flag = nafs_earth_kyoshi_done }
					}
					character_event = { id = NAFS.305 }
				}
			}
		}
	}
}
