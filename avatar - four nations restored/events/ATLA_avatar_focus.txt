# Cooldown cleanup and modest long-term progression tracking.
character_event = {
    id = avatar_focus.60
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        clr_character_flag = avatar_focus_activity_cooldown
        if = {
            limit = { NOT = { check_variable = { which = avatar_focus_activities_completed value >= 1 } } }
            set_variable = { which = avatar_focus_activities_completed value = 0 }
        }
        change_variable = { which = avatar_focus_activities_completed value = 1 }
        if = {
            limit = { check_variable = { which = avatar_focus_activities_completed value = 3 } NOT = { has_character_flag = avatar_focus_veteran } }
            set_character_flag = avatar_focus_veteran
            character_event = { id = avatar_focus.70 }
        }
        # --- Focus rank progression -------------------------------------
        # Completed undertakings advance the Rank; thresholds are 3 / 7 /
        # 12 / 18. Rank 5 is the summit of the path and grants a mastery
        # trait. avatar_focus.205 dispatches to the right event.
        if = {
            limit = { NOT = { check_variable = { which = avatar_focus_progress value >= 1 } } }
            set_variable = { which = avatar_focus_progress value = 0 }
        }
        if = {
            limit = { NOT = { check_variable = { which = avatar_focus_rank value >= 1 } } }
            set_variable = { which = avatar_focus_rank value = 1 }
        }
        change_variable = { which = avatar_focus_progress value = 1 }
        if = {
            limit = {
                check_variable = { which = avatar_focus_progress value = 18 }
                NOT = { check_variable = { which = avatar_focus_rank value = 5 } }
            }
            set_variable = { which = avatar_focus_rank value = 5 }
            character_event = { id = avatar_focus.205 }
        }
        else_if = {
            limit = {
                check_variable = { which = avatar_focus_progress value = 12 }
                NOT = { check_variable = { which = avatar_focus_rank value = 4 } }
            }
            set_variable = { which = avatar_focus_rank value = 4 }
            character_event = { id = avatar_focus.205 }
        }
        else_if = {
            limit = {
                check_variable = { which = avatar_focus_progress value = 7 }
                NOT = { check_variable = { which = avatar_focus_rank value = 3 } }
            }
            set_variable = { which = avatar_focus_rank value = 3 }
            character_event = { id = avatar_focus.205 }
        }
        else_if = {
            limit = {
                check_variable = { which = avatar_focus_progress value = 3 }
                NOT = { check_variable = { which = avatar_focus_rank value = 2 } }
            }
            set_variable = { which = avatar_focus_rank value = 2 }
            character_event = { id = avatar_focus.205 }
        }
    }
}
