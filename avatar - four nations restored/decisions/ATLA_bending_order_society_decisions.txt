########################################################################
# ATLA Bending Orders - recruitment and sparring
#
# The two things the Agni Kai has and the Waterbending Order and the
# Earthbending Society did not:
#
#   wbo_/ebo_recruit_member       (rank 2, targeted) bring a new bender in
#   wbo_/ebo_spar_fellow_member   (any rank, targeted) spar with a member
#
# Targeted decisions follow the mod's usual convention: the right-clicked
# character is the implicit scope (ROOT), FROM is the decision taker.
#
# The sparring bout is settled inside the decision rather than in an event
# chain. Both characters get their own result letter from
# events/ATLA_bending_order_society_events.txt, so nothing has to carry a
# scope from one character's event to another.
########################################################################

targeted_decisions = {

	wbo_recruit_member = {
		ai_check_interval = 24

		}

########################################################################
# The rank ladder.
#
# Each order promotes on its own terms. The rank-up button in the Society
# view points at these decisions (society_rank_up_decision in
# common/societies/ATLA_bending_orders.txt).
#
# A member climbs from the first rank to the third by earning the order's
# regard and proving the order's virtue:
#   Water - learning, and time served
#   Earth - martial skill, and time served
# The fourth rank is leadership, and is not asked for: it is reached through
# the society's own standing.
########################################################################

society_decisions = {

	wbo_request_rank_up = {
		only_playable = yes
		hide_in_decisions_list = yes
		ai_check_interval = 24

		potential = {
			society_member_of = waterbending_order
			society_rank < 3
			days_in_society > 365
		}

		allow = {
			custom_tooltip = {
				text = wbo_rank_up_currency_tt
				OR = {
					AND = {
						society_rank = 1
						society_currency >= 750
					}
					society_currency >= 1000
				}
			}
			custom_tooltip = {
				text = wbo_rank_up_time_tt
				days_at_current_society_rank >= 730
			}
			custom_tooltip = {
				text = wbo_rank_up_skill_tt
				OR = {
					AND = {
						society_rank = 1
						learning >= 10
					}
					learning >= 14
				}
			}
		}

		effect = {
			if = {
				limit = { society_rank = 1 }
				change_society_currency = -750
			}
			if = {
				limit = { society_rank = 2 }
				change_society_currency = -1000
			}
			society_rank_up = 1
		}

		ai_will_do = {
			factor = 0.5
