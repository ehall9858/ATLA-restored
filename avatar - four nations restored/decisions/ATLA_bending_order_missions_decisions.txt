########################################################################
# ATLA Bending Orders - mission decisions
#
# A mission is handed out by events/ATLA_bending_order_missions.txt. The
# task itself is completed here: a decision appears while the quest is
# active, and taking it does the work, pays the Order, and closes the quest.
#
#  Waterbending Order
#    wbo_mission_tend_the_sick      heal a wounded or ailing courtier
#    wbo_mission_build_healing_hut  found a healing hut in your capital
#    wbo_mission_bring_new_voice    bring a waterbender into the Order
#    wbo_mission_set_aside          put the task down
#
#  Earthbending Society
#    ebo_mission_raise_the_wall     raise a wall at your capital
#    ebo_mission_stand_the_trial    stand the Society's trial of endurance
#    ebo_mission_bring_new_pillar   bring an earthbender into the Society
#    ebo_mission_set_aside          put the task down
########################################################################

decisions = {

	# ------------------------------------------------------------- water: heal
	wbo_mission_tend_the_sick = {
		only_playable = yes
		ai_check_interval = 6
		is_high_prio = yes

		potential = {
			has_quest = quest_wbo_tend_the_sick
			any_courtier = {
				OR = {
					trait = wounded
					trait = stressed
					trait = depressed
				}
			}
		}

		allow = {
			wealth >= 20
			prisoner = no
			NOT = { trait = incapable }
		}

		effect = {
			wealth = -20
			random_courtier = {
				limit = {
					OR = {
						trait = wounded
						trait = stressed
						trait = depressed
					}
				}
				if = {
					limit = { trait = wounded }
					remove_trait = wounded
				}
				else_if = {
					limit = { trait = stressed }
					remove_trait = stressed
				}
				else = {
					remove_trait = depressed
				}
			}
			piety = 40
			prestige = 20
			add_society_currency_major_effect = yes
			clr_quest = quest_wbo_tend_the_sick
			set_character_flag = wbo_mission_done_sick
			character_event = { id = atla_bending_missions.30 days = 3 }
		}

		ai_will_do = {
			factor = 1
		}
	}

		}

		}
