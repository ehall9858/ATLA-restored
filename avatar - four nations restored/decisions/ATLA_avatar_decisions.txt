	# --- Withdrawal from Avatar duties ----------------------------------------
	# Withdrawal is a voluntary resignation from the Avatar role. While the
	# flag is set the Avatar cannot open the interference panel, their peace
	# terms are ignored, and NAE.229 stops advertising them as available.
	# Staying withdrawn for 11 continuous years gets them declared unworthy
	# (NAE.233 -> NAE.106). The AI only withdraws when its personality has
	# genuinely broken down.
	avatar_withdraw = {
		is_high_prio = yes
		ai_will_do = { factor = 1 }

		potential = {
			is_revealed_avatar_trigger = yes
			NOT = { has_character_flag = withdrawn_avatar }
			OR = {
				ai = no
				calc_true_if = {
					amount = 3
					trait = depressed
					trait = slothful
					trait = cynical
					trait = drunkard
					trait = cactusaddict
					trait = envious
					trait = shy
				}
			}
		}

		allow = {
			is_incapable = no
			NOT = { prisoner = yes }
		}

		effect = {
			any_player = { character_event = { id = NAE.516 } } #Declares the Avatar has gone into withdrawal

		}
	}

	move_avatar = {
		is_high_prio = yes
		only_playable = yes
		filter = independent_rulers
		ai = no
		
		from_potential = {
			is_landed = no
			is_avatar_trigger = yes
		}
		
		potential = {
			# The Avatar roams the whole world, so a destination does not
			# have to sit inside the Avatar's diplomatic range.
			# is_within_diplo_range = FROM
			is_landed = yes
			# you're not already a tributary
			FROM = {
				NOR = {
					is_tributary = {
						type = unrevealed_avatar_tributary
						suzerain = PREV
					}
					is_tributary = {
						type = avatar_tributary
						suzerain = PREV
					}
				}
			}
		}
		
		allow = {
			# The Avatar goes where the world needs them. Court opinion,
			# old feuds, being mid-journey and the distance to the target
			# no longer bar the road: if the option is offered, it can be
			# taken. (This also fixed travelling while the Avatar holds no
			# capital holding, where the old distance check had no origin
			# and so could never pass.)
			always = yes
		}
		effect = {
			# Store the clicked ruler as the journey destination BEFORE the
			# Avatar is asked: avatar_journey_effect and new_avatar_journey.50
			# both read event_target:avatar_journey_target, and nothing else
			# ever saved it - so accepting the move silently did nothing.
			clear_event_target = avatar_journey_target
			save_event_target_as = avatar_journey_target
			custom_tooltip = {
				text = move_avatar_tt
				FROM = {
					character_event = {
						id = new_avatar_journey.10
					}
				}
			}
		}
	}

	# Only one crisis at a time: avatar_crisis_active blocks the rest.
	declare_crisis_war = {
		is_high_prio = yes

		ai = no

			OR = {

						has_province_modifier = colony_3
					}
				}
				AND = {
					has_law = slavery_legal
					realm_size >= 40

			}
		}

		allow = {
			FROM = {
				is_adult = yes

				NOT = { prisoner = yes }
				NOT = { has_character_flag = avatar_crisis_war_used }

			save_event_target_as = crisis_target
			FROM = { character_event = { id = avatar_crisis.1 } }
		}
	}

	# Right-click any landed ruler to declare them a Villain of the world.
	# From_potential = you (the Avatar). potential/allow/effect = the ruler you
	# right-clicked; FROM = you. Confirmation (and the duel challenge) runs
	# through avatar_villain.2.
	#
	# The ruler must actually be a villain candidate (is_villain_candidate_trigger
	# - the same test the yearly warning uses), so the declaration is earned
	# rather than a free source of prestige. There is still no cooldown between
	# declarations, so a player Avatar can work through a list of them. The AI
	# never gets this decision (ai = no); it only ever acts through the rare
	# auto-warning (avatar_villain.1), which keeps its own long cooldown. The
	# one thing still blocked is a new declaration while an Avatar-villain duel
	# is unresolved, which would otherwise overwrite the running duel's
	# combatant_2 / villain_candidate.
	denounce_villain = {
		is_high_prio = yes
		filter = all
		ai = no

		from_potential = {
			is_avatar_trigger = yes
			trait = fullyrealisedavatar
		}

		potential = {
			is_landed = yes
			is_adult = yes
			NOT = { character = FROM }
			NOT = { trait = villain }
			NOT = { has_character_modifier = villain_reprieved }
		}

		allow = {
			# The full eligibility test lives in allow, not potential: potential
			# runs for every ruler in the world each time the menu is opened,
			# and this one walks their provinces.
			custom_tooltip = {
				text = villain_candidate_tt
				is_villain_candidate_trigger = yes
			}
			FROM = {
				is_adult = yes
				is_incapable = no
				NOT = { prisoner = yes }
				# No villain cooldown: the player may chain declarations.
				# But never start a new stake mid-duel.
				NOT = { has_character_flag = avatar_villain_duel }
			}
		}

		effect = {
			save_event_target_as = villain_candidate
			FROM = { character_event = { id = avatar_villain.2 } }
		}
	}
