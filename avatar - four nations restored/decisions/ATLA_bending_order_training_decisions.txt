########################################################################
# ATLA Bending Orders - study, mastery and instruction
#
# The ladder is written to the shape of the Brotherhood of Agni Kai:
#
#   rank 1   practice                the forms, and sparring
#   rank 2   meditate, train         climb the bending tiers
#   rank 3   teach, instruct         teach the young
#   rank 4   convene, sub-art        mastery of a sub-bending art
#
# The sub-bending arts are the privilege of the head of the order, exactly as
# bak_learn_new_bending is the privilege of the Agni Kai's grandmaster. A
# character is taught one art, once, and never by an ordinary member.
#
# Nothing in this file grants prestige. The orders are a source of skill, not
# a source of renown.
#
# Targeted decisions follow the mod's usual convention: the right-clicked
# character is the implicit scope (ROOT), FROM is the decision taker.
########################################################################

targeted_decisions = {

	# ---------------------------------------------------------------- water
	wbo_instruct_bender = {
		ai_check_interval = 12
		filter = realm
		ai_target_filter = home_court
		is_in_society = yes

		potential = {
			NOT = { trait = incapable }
			is_in_society = no
			is_adult = no
			OR = {
				trait = waterbender
				trait = avatar_waterbender
			}
			NOT = { has_character_flag = received_wbo_education_before }
			is_top_tier_bender_trigger = no
		}

		from_potential = {
			society_rank >= 3
			society_member_of = waterbending_order
			is_adult = yes
			NOT = { trait = incapable }
		}

		allow = {
			FROM = {
				has_society_currency_major_trigger = yes
				is_inaccessible_trigger = no
				prisoner = no
				in_command = no
			}
			is_inaccessible_trigger = no
			prisoner = no
		}

		effect = {
			FROM = {
				detract_society_currency_major_effect = yes
			}
			ROOT = {
				set_character_flag = received_wbo_education_before
				if = {
					limit = { is_bender_trigger = no }

					upgrade_bending_effect = yes
				}
				hidden_tooltip = {
					character_event = { id = atla_bending_order.40 }
