upgrade_bending_effect = { ##Just adding new traits sufficient, since opposites already established
		## This is the "being taught" ladder - the educator events, the bending orders,
		## the Agni Kai. The steps below powerful are ordinary; the step into master is
		## rare and only for a formidable bender; and a master who is being pushed by
		## real teachers has a long shot at going beyond the rank entirely.
		## See common/scripted_triggers/10_atla_bending_rarity_triggers.txt.
		## (train_your_bending does NOT use this effect - it tops out at powerful.)
		trigger_switch = {
			on_trigger = trait
			incompetent_bender = { add_trait = competent_bender }
			competent_bender = { add_trait = powerful_bender }
			powerful_bender = { try_become_master_bender_effect = yes }
			master_bender = { try_become_legendary_bender_effect = yes }
			fallback = { }
		}
}

set_duel_variables_effect = {
	set_variable = { which = myHP value = 30 } # SET UP HP
	set_variable = { which = myDEF value = 3 } # SET UP DEF
	set_variable = { which = myATK value = 10 } # SET UP ATK

	if = { # Avatar = +20 HP, +2 DEF, +3 ATK
		limit = { trait = avatar }
		change_variable = { which = myHP value = 20 }
		change_variable = { which = myDEF value = 2 }
		change_variable = { which = myATK value = 3 }

		change_variable = { which = myDEF value = 1 }

		limit = { trait = canon_character }

		change_variable = { which = myATK value = 1 }
		change_variable = { which = myDEF value = 1 }

	set_variable = { which = myHPcap which = myHP } # SET UP HP CAP
}

		change_variable = { which = vSHO value = 3 }

		change_variable = { which = vSHO value = 7 }

		change_variable = { which = vPAC value = 10 }
		change_variable = { which = vSHO value = 10 }
		change_variable = { which = vDEF value = 10 }
	}
	if = { # Quick
		limit = { trait = quick }
		change_variable = { which = vPAC value = 2 }
		change_variable = { which = vSHO value = 2 }
	}
	if = { # Shrewd
		limit = { trait = shrewd }
		change_variable = { which = vPAC value = 1 }
		change_variable = { which = vSHO value = 1 }
	}
	if = { # Slow
		limit = { trait = slow }
		subtract_variable = { which = vPAC value = 1 }
		subtract_variable = { which = vSHO value = 1 }
	}
	if = { # One Legged
		limit = { trait = one_legged }
		subtract_variable = { which = vPAC value = 3 }
		subtract_variable = { which = vSHO value = 2 }
	}
	if = { # Wounded
		limit = { trait = wounded }
		subtract_variable = { which = vPAC value = 1 }
		subtract_variable = { which = vSHO value = 1 }
		subtract_variable = { which = vDEF value = 1 }
	}
	if = { # Weak
	limit = { trait = weak }
		subtract_variable = { which = vPAC value = 1 }
		subtract_variable = { which = vSHO value = 1 }
		subtract_variable = { which = vDEF value = 1 }
	}
	if = { # Maimed
		limit = { trait = maimed }
		subtract_variable = { which = vPAC value = 2 }
		subtract_variable = { which = vSHO value = 2 }
		subtract_variable = { which = vDEF value = 2 }
	}
	if = { # Diseased
		limit = { has_disease_trigger = yes }
		subtract_variable = { which = vPAC value = 1 }
		subtract_variable = { which = vSHO value = 1 }
		subtract_variable = { which = vDEF value = 1 }
	}
	if = { # One Handed
		limit = { trait = one_handed }
		subtract_variable = { which = vSHO value = 2 }
	}
	if = { # Strong
		limit = { trait = strong }
		change_variable = { which = vDEF value = 1 }
	}
	if = { # Robust
		limit = { trait = robust }
		change_variable = { which = vDEF value = 1 }
	}

	export_to_variable = { which = addPAC value = combat_rating }
	divide_variable = { which = addPAC value = 3 }
	change_variable = { which = vPAC which = addPAC }

	# Setup shooting

	export_to_variable = { which = addSHO value = combat_rating }
	divide_variable = { which = addSHO value = 2 }
	change_variable = { which = vSHO which = addSHO }

	# setup defending
	export_to_variable = { which = addDEF value = combat_rating }
	divide_variable = { which = addDEF value = 4 }
	change_variable = { which = vDEF which = addDEF }

	# Setup phy
	export_to_variable = { which = vPHY value = combat_rating }

	# final rating
	set_variable = { which = final_rating value = 0 } # pace

	set_variable = { which = vFPAC value = 0 } # pace
	set_variable = { which = vFSHO value = 0 } # shooting
	set_variable = { which = vFDEF value = 0 } # defending
	set_variable = { which = vFPHY value = 0 } # physical

	change_variable = { which = vFPAC which = vPAC } # pace
	change_variable = { which = vFSHO which = vSHO } # shooting
	change_variable = { which = vFDEF which = vDEF } # defending
	change_variable = { which = vFPHY which = vPHY } # physical

	multiply_variable = { which = vFPAC value = 0.3 }
	multiply_variable = { which = vFSHO value = 0.3 }
	multiply_variable = { which = vFDEF value = 0.2 }
	multiply_variable = { which = vFPHY value = 0.2 }

	change_variable = { which = final_rating which = vFPAC }
	change_variable = { which = final_rating which = vFSHO }
	change_variable = { which = final_rating which = vFDEF }
	change_variable = { which = final_rating which = vFPHY }
}

# The world's answer to a declaration. Call only on the Avatar's scope, with
# event_target:villain_candidate set to the declared Villain and
# event_target:villain_declarer set to the Avatar. Rulers turn on the villain
# and remember who named them; the villain's own vassals lose faith in them.
declared_villain_reaction_effect = {
	any_independent_ruler = {
		limit = { NOT = { character = event_target:villain_candidate } }
		opinion = { modifier = opinion_declared_villain who = event_target:villain_candidate years = 10 }
		opinion = { modifier = opinion_avatar_champion who = event_target:villain_declarer years = 10 }
	}
	event_target:villain_candidate = {
		any_vassal = {
			opinion = { modifier = opinion_villain_liege who = event_target:villain_candidate years = 5 }
		}
	}
}

# Destroy every ducal-or-better title a defeated Villain holds - all of them.
# A big AI "nation" is often nothing but a fat duchy, and an empire can hold
# several kingdoms besides; leaving any of them standing leaves the villain's
# power base intact.
#
# One sweep is not enough. Destroying the primary title ends an
# any_demesne_title iteration, so the sweep is repeated until the demesne has
# nothing ducal or above left. The guard variable caps the loop, so a title
# the engine refuses to destroy can never hang the game. Counties are
# deliberately spared: the point is to break the realm, not to unland every
# holding.
#
# Call inside the villain's scope.
dismantle_villain_realm_effect = {
	hidden_tooltip = {
		set_variable = { which = atla_villain_dismantle_guard value = 0 }
	}
	while = {
		limit = {
			check_variable = { which = atla_villain_dismantle_guard value < 40 }
			any_demesne_title = { higher_tier_than = COUNT }
		}
		hidden_tooltip = {
			change_variable = { which = atla_villain_dismantle_guard value = 1 }
		}
		any_demesne_title = {
			limit = { higher_tier_than = COUNT }
			destroy_landed_title = THIS
		}
	}
	hidden_tooltip = {
		remove_variable = atla_villain_dismantle_guard
	}
}

# The four fates of a defeated Villain. Each runs in the villain's scope; ROOT

# villain forfeited) so the two paths cannot drift apart again.

villain_stake_kill_and_dismantle_effect = {

	abdicate = yes

}
