########################################################################
# ATLA Ambition Pack
#
# Native ambitions written for the elis_fork systems (is_bender_trigger,
# is_top_tier_bender_trigger, sub-bending traits, is_avatar_trigger).
#
# Purpose: give the general cast goals of their own, so AI characters are
# driven by something more than the Avatar's story. These are NOT for the
# Avatar, who already has a bespoke scripted life.
########################################################################

#  BENDING AMBITIONS   #
########################

# Rise from an ordinary bender to a true master of the art.
obj_master_your_bending = {
	type = character

	potential = {
		is_bender_trigger = yes
		NOT = { is_top_tier_bender_trigger = yes }
		NOT = { is_avatar_trigger = yes }
	}

	allow = {
		is_adult = yes
		is_incapable = no
		prisoner = no
	}

	chance = {
		factor = 60

		modifier = {
			factor = 2.0
			trait = ambitious
		}
		modifier = {
			factor = 1.5
			trait = diligent
		}
		modifier = {
			factor = 1.25
			trait = patient
		}
		modifier = {
			factor = 1.25
			trait = brave
		}
		modifier = {
			factor = 0.5
			trait = slothful
		}
		modifier = {
			factor = 0.25
			trait = content
		}
	}

	success = {
		is_top_tier_bender_trigger = yes
	}

	abort = {
		OR = {
			NOT = { is_bender_trigger = yes }
			is_incapable = yes
			prisoner = yes
		}
	}

	abort_effect = {
	}

	effect = {
		prestige = 150
		change_martial = 1
		set_character_flag = achieved_bending_mastery
	}
}

# The warrior's ladder as an ambition. Experience (atla_warrior_xp) is earned
# in battle (ATLAW.30-31) and through the yearly training event (ATLAW.32);
# the thresholds live in atla_warrior_check_rank_effect. Success is simply
# reaching the top of the non-bender ladder.
obj_master_my_weapon = {
	type = character

	potential = {
		NOT = { is_any_bender_trigger = yes }
		NOT = { trait = atla_grandmaster }
	}

	allow = {
		is_adult = yes
		is_incapable = no
		prisoner = no
	}

	chance = {
		factor = 40

		modifier = {
			factor = 2.0
			trait = ambitious
		}
		modifier = {
			factor = 1.5
			trait = diligent
		}
		modifier = {
			factor = 1.25
			trait = brave
		}
		modifier = {
			factor = 1.25
			trait = wroth
		}
		modifier = {
			factor = 1.5
			martial = 15
		}
		modifier = {
			factor = 0.5
			trait = slothful
		}
		modifier = {
			factor = 0.25
			trait = content
		}
	}

	success = {
		trait = atla_grandmaster
	}

	abort = {
		OR = {
			is_incapable = yes
			prisoner = yes
			is_any_bender_trigger = yes
		}
	}

	abort_effect = {
		set_variable = { which = atla_warrior_xp value = 0 }
	}

	effect = {
		prestige = 250
	}
}
