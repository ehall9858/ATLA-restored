# Bending orders: written from scratch, structured like the Brotherhood of Agni Kai.
#
# The two societies are now proper orders rather than placeholders.
# Each has a named ladder of four ranks, a rank-up decision in the Society view,
# a recruitment decision, a sparring decision, and a set of society missions
# (see decisions/ATLA_bending_order_society_decisions.txt and
# events/ATLA_bending_order_missions.txt).
#
# Sub-bending arts are taught only to the head of the order (rank 4), exactly as
# bak_learn_new_bending is the privilege of the Agni Kai's grandmaster.
#
# Rank ladder (loc keys waterbending_order_rank_N_*, earthbending_order_rank_N_*):
#   Water: Still Water, Flowing Water, Deep Water, The Turning Tide
#   Earth: Stone, Boulder, Crag, Mountain

waterbending_order = { # The Order of Flowing Water
	primary_attribute = learning
	opinion_to_other_members = +5
	opinion_per_rank_above = +10
	sound = monastic_orders_interface
	society_influence = no

	}

		custom_tooltip = waterbending_order_rank_3_duty
	}

	society_rank = { # The Turning Tide

		limit = 1
		startup_limit = 1
		modifier = {
			monthly_character_piety = 0.5
			health = 1.0
		}
		decisions = {
			wbo_convene
			wbo_learn_subbending
		}
		custom_tooltip = waterbending_order_rank_4_duty
	}

	monthly_currency_gain = {
		name = waterbending_order_currency

		per_attribute = {
			name = learning
			value = 0.5
		}
		has_trait = {
			trait = legendary_bender
			value = 5
		}
		has_trait = {
			trait = master_bender
			value = 3
		}
		has_trait = {
			trait = powerful_bender
			value = 3
		}
		has_trait = {
			trait = competent_bender
			value = 2
		}
		has_trait = {
			trait = incompetent_bender
			value = 1
		}
		# The Order values what water is for: mending what is broken.
		has_trait = {
			trait = master_healer
			value = 3
		}
		has_trait = {
			trait = healer
			value = 2
		}
		has_trait = {
			trait = positive_jing
			value = 1
		}
		has_trait = {
			trait = patient
			value = 1
		}
		has_trait = {
			trait = diligent
			value = 1
		}
		has_trait = {
			trait = kind

		has_trait = {
			trait = content
			value = 1

		society_rank = {
			rank = 4
			value = 10
		}
	}

	member_score_per_attribute = 3

	member_score = {

			trait = legendary_bender
			factor = 5

		modifier = {
			trait = diligent
			factor = 1.1
		}

		trigger = {
			ai = yes
			age = 16

			is_in_society = no
		}
	}
}
