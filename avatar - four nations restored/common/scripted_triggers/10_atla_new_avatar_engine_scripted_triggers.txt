is_suitable_mother_of_avatar_trigger = {
    is_female = yes
    is_adult = yes
    is_married = yes
    # Noble branch only: both parents must already belong to a dynasty, so the
    # child really is born into a noble house. Without this the "noble" roll
    # could land on a lowborn mother and produce a dynasty-less child anyway.
    is_lowborn = no
    spouse_even_if_dead = {
        is_lowborn = no
    }

    OR = {
        liege = {
            higher_real_tier_than = BARON
        }
        top_liege = {
            higher_real_tier_than = COUNT
        }
    }

    OR = {
        AND = {
            has_global_flag = avatar_fire
            is_suitable_mother_of_avatar_fire_trigger = yes
        }
        AND = {
            has_global_flag = avatar_air
            is_suitable_mother_of_avatar_air_trigger = yes
        }
        AND = {
            has_global_flag = avatar_water
            is_suitable_mother_of_avatar_water_trigger = yes
        }
        AND = {
            has_global_flag = avatar_earth
            is_suitable_mother_of_avatar_earth_trigger = yes
        }
    }
}

# Same as above, but for a POOR family: lowborn, unmarried (her husband is
# generated lowborn at birth time), and herself a bender of the right
# element - no rank requirement.
# NOTE: the lowborn birth path no longer uses this as a gate - it generates
# both parents from scratch, so it never depends on an existing family. Kept
# only in case something wants to detect "a world where lowborn benders exist".
is_suitable_lowborn_mother_of_avatar_trigger = {
    is_female = yes
    is_adult = yes
    is_married = no
    is_lowborn = yes

    OR = {
        AND = {
            has_global_flag = avatar_fire
            trait = firebender
        }
        AND = {
            has_global_flag = avatar_air
            trait = airbender
        }
        AND = {
            has_global_flag = avatar_water
            trait = waterbender
        }
        AND = {
            has_global_flag = avatar_earth
            trait = earthbender
        }
    }
}

is_hostile_to_avatar_trigger = {
	custom_tooltip = { text = hostile_to_avatar_tt }
	OR = {
		AND = {
			is_heretic = yes
			NOT = { religion = followers_of_kyoshi }
		}
		religion = way_of_chin
		religion = equalism
		religion = barbarism
		has_religion_feature = religion_feature_old_barbarism #This is the 'We're gonna extinguish the waterbenders' doctrine
		has_religion_feature = religion_feature_new_barbarism #This is the 'We are going to sacrifice people en-masse to dark gods' doctrine
		religion = fire_nation_imperialism
		has_law = slavery_legal
		any_rival = { is_revealed_avatar_trigger = yes }
		any_current_enemy = { is_revealed_avatar_trigger = yes } #Checks the Avatar isn't currently at war with them
	}
}

has_tyrannical_personality_trigger = { #These numbers are chosen specifically to ensure certain personality combinations trip it. Don't adjust them!
	custom_tooltip = { text = tyrannical_personality_tt } #Has tyrannical personality
	calc_true_if = {
		amount = 3
		ai_greed = 25 #Greed is more a degeneracy check
		ai_ambition = 22
		ai_honor <= -15 #Remember that 'honor' is closer to 'trustworthyness' than anything else. This isn't a check for raw niceness.
		ai_rationality <= -30 #Tyrants are fine, MAD Tyrants are all the rage!
	}
}
